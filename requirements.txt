# Mini Project 2 — dependency reference.
#
# ---------------------------------------------------------------------------
# READ THIS FIRST: conda is the supported install route.
# ---------------------------------------------------------------------------
#
# geopandas needs GDAL, GEOS and PROJ underneath. pip wheels for these are built
# against whatever versions were current when each package was uploaded, so you
# can end up with a wheel built for one GDAL and a pyproj built for another —
# and "import geopandas" fails at import time with a version conflict that is
# genuinely fiddly to unpick.
#
# conda-forge solves those three together as a set. So install this way:
#
#     conda create -n mp2 -c conda-forge python=3.11 \
#         geopandas fiona shapely pyproj \
#         pandas numpy scikit-learn kmodes matplotlib jupyterlab -y
#     conda activate mp2
#     python -m ipykernel install --user --name mp2 --display-name "Python (mp2)"
#
# The pins below are the versions this project was built and checked against.
# Conda will resolve to compatible versions itself; use this file as the record
# of what "known good" means, and if you install with pip instead, run
# "pip freeze > requirements.lock.txt" and commit that.
#
